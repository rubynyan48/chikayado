const searchParams = new URLSearchParams(window.location.search); let diffYear, diffMonth, diff; // get start slide number if (searchParams.has('y')) { diffYear = (searchParams.get('y') - 2023) * 12; diffMonth = searchParams.get('m') - 9; } else { let today = new Date(); diffYear = (today.getFullYear() - 2023) * 12; diffMonth = today.getMonth() + 1 - 9; } diff = diffYear + diffMonth; // splide document.querySelectorAll('.splide').forEach((splideElement) => { let splide = new Splide(splideElement, { drag: true, width: '100%', fixedWidth: 80, fixedHeight: 80, perPage: 7, perMove: 7, gap: 20, padding: { left: 10, right: 10 }, focus: 0, pagination: false, start: diff, breakpoints: { 749: { width: '100%', perPage: 3, perMove: 3, }, }, }).mount(); splide.on('click', function (Slide) { let url, year, month; url = location.protocol + '//' + location.hostname + location.pathname; year = Slide.slide.dataset.year; month = Slide.slide.dataset.month; location.href = url + '?y=' + year + '&m=' + month; }); });
